    private string _v5AlertFilter = "ALL";

    private void SetV5AlertFilter(string filter)
    {
        _v5AlertFilter = filter;
        StateHasChanged();
    }

    private List<AlertRow> V5FilteredAlerts => _v5AlertFilter switch
    {
        "CRITICAL" => _allAlerts.Where(a => a.Level == "red").ToList(),
        "HIGH" => _allAlerts.Where(a => a.Level == "amber").ToList(),
        "MEDIUM" => _allAlerts.Where(a => a.Level == "yellow").ToList(),
        _ => _allAlerts.ToList()
    };

    // Crisis-prioritized actions: red/amber first, then green
    private List<ActionRow> V5CrisisActions => FilteredActions
        .OrderBy(a => a.Level == "red" ? 0 : a.Level == "amber" ? 1 : 2)
        .ThenBy(a => a.Deadline)
        .ToList();
