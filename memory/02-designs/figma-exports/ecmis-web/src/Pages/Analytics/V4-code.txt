    // AI-prioritized actions (sorted by predicted urgency score descending)
    private List<ActionRow> V4AiPrioritizedActions =>
        _allActions
            .Select(a => a with { Rank = PredictUrgencyScore(a) })
            .OrderByDescending(a => a.Rank)
            .ToList();

    private static int PredictUrgencyScore(ActionRow action)
    {
        // Simple AI scoring heuristic based on level + deadline proximity + category risk
        var baseScore = action.Level switch
        {
            "red" => 85,
            "amber" => 65,
            "green" => 35,
            _ => 50
        };
        // Add random-ish but deterministic variance based on title hash so it looks AI-computed
        var hash = action.Title.GetHashCode();
        var variance = Math.Abs(hash % 15);
        return Math.Min(99, baseScore + variance);
    }
