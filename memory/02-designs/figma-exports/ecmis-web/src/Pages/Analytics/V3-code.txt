// ── V3 State ──
private int _v3TimeSlider = 6;
private string _v3SortColumn = "score";
private bool _v3SortAsc = false;
private HashSet<string> _v3SelectedProvinces = new();
private string? _v3ActionRegionFilter = null;

// ── V3 Computed: Sorted Provinces ──
private List<ProvinceScore> SortedProvinces
{
    get
    {
        var list = _byProvince.ToList();
        list = _v3SortColumn switch
        {
            "name" => _v3SortAsc ? list.OrderBy(p => p.Name).ToList() : list.OrderByDescending(p => p.Name).ToList(),
            "score" => _v3SortAsc ? list.OrderBy(p => p.Score).ToList() : list.OrderByDescending(p => p.Score).ToList(),
            "rank" => _v3SortAsc ? list.OrderBy(p => p.Index).ToList() : list.OrderByDescending(p => p.Index).ToList(),
            _ => list
        };
        return list;
    }
}

// ── V3 Computed: Filtered Actions by Region Chip ──
private List<ActionRow> V3FilteredActions
{
    get
    {
        var actions = FilteredActions;
        if (string.IsNullOrEmpty(_v3ActionRegionFilter))
            return actions;
        // Map action owner to region for demo purposes (deterministic fake mapping)
        return actions.Where(a => ActionOwnerToRegion(a.Owner) == _v3ActionRegionFilter).ToList();
    }
}

// ── V3 Helper: Map owner to region for demo filtering ──
private static string ActionOwnerToRegion(string owner)
{
    return owner switch
    {
        "ป.ป.ท." => "ภาคกลาง",
        "ป.ป.ช." => "ภาคเหนือ",
        "DSI" => "ตะวันออก",
        "สตง." => "ภาคใต้",
        _ => "ตะวันออกเฉียงเหนือ"
    };
}

// ── V3 Helper: Sort indicator text ──
private string SortIndicator(string column)
{
    if (_v3SortColumn != column) return "";
    return _v3SortAsc ? "▲" : "▼";
}

// ── V3 Helper: Region color for radar chart legend ──
private static string V3RegionColor(string regionName)
{
    return regionName switch
    {
        "ภาคเหนือ" => "#2dd4bf",
        "ตะวันออกเฉียงเหนือ" => "#34d399",
        "ภาคกลาง" => "#14b8a6",
        "ตะวันออก" => "#10b981",
        "ภาคใต้" => "#059669",
        _ => "#2dd4bf"
    };
}

// ── V3 Interaction: Sort province table ──
private void SortProvinceTable(string column)
{
    if (_v3SortColumn == column)
        _v3SortAsc = !_v3SortAsc;
    else
    {
        _v3SortColumn = column;
        _v3SortAsc = column == "name"; // default asc for name, desc for score/rank
    }
}

// ── V3 Interaction: Toggle province selection for comparison ──
private void ToggleProvinceSelection(string provinceName)
{
    if (_v3SelectedProvinces.Contains(provinceName))
        _v3SelectedProvinces.Remove(provinceName);
    else if (_v3SelectedProvinces.Count < 3)
        _v3SelectedProvinces.Add(provinceName);
    // If already 3 selected, ignore new selection (or you could swap logic)
}

// ── V3 Interaction: Set action region filter ──
private void SetV3ActionRegionFilter(string? region)
{
    _v3ActionRegionFilter = region;
}
