// ── V4: Predictive Analytics — AI Forecast Chart with Confidence Bands ──

window.initForecastChart = function (canvasId, labels, forecastData, upperBound, lowerBound, actualData, isDark) {
    if (window._analyticsCharts[canvasId]) {
        window._analyticsCharts[canvasId].destroy();
        delete window._analyticsCharts[canvasId];
    }
    const canvas = document.getElementById(canvasId);
    if (!canvas) return;

    isDark = isDark !== false;
    const gridColor  = isDark ? 'rgba(255,255,255,0.05)' : 'rgba(13,27,62,0.07)';
    const tickColor  = isDark ? 'rgba(255,255,255,0.4)'  : 'rgba(13,27,62,0.5)';
    const tooltipBg  = isDark ? 'rgba(8,12,28,0.95)'     : 'rgba(255,255,255,0.97)';
    const tooltipTxt = isDark ? '#fff'                    : '#1a3575';

    // Build gradient for confidence band fill
    const ctx = canvas.getContext('2d');
    const gradientFill = ctx.createLinearGradient(0, 0, 0, 300);
    gradientFill.addColorStop(0, 'rgba(139, 92, 246, 0.22)');
    gradientFill.addColorStop(1, 'rgba(236, 72, 153, 0.04)');

    window._analyticsCharts[canvasId] = new Chart(ctx, {
        type: 'line',
        data: {
            labels: labels,
            datasets: [
                // Upper bound (dashed, faint)
                {
                    label: 'Upper Bound (95%)',
                    data: upperBound,
                    borderColor: 'rgba(139, 92, 246, 0.35)',
                    backgroundColor: 'transparent',
                    fill: false,
                    tension: 0.35,
                    borderWidth: 1.5,
                    borderDash: [4, 4],
                    pointRadius: 0,
                    pointHitRadius: 0,
                    order: 3
                },
                // Lower bound (dashed, faint) — fill to upper
                {
                    label: 'Lower Bound (95%)',
                    data: lowerBound,
                    borderColor: 'rgba(139, 92, 246, 0.35)',
                    backgroundColor: gradientFill,
                    fill: '-1',
                    tension: 0.35,
                    borderWidth: 1.5,
                    borderDash: [4, 4],
                    pointRadius: 0,
                    pointHitRadius: 0,
                    order: 4
                },
                // Forecast main line
                {
                    label: 'AI Forecast',
                    data: forecastData,
                    borderColor: '#a855f7',
                    backgroundColor: 'rgba(168, 85, 247, 0.08)',
                    fill: false,
                    tension: 0.4,
                    borderWidth: 2.5,
                    pointRadius: 3,
                    pointBackgroundColor: '#d946ef',
                    pointBorderColor: '#fff',
                    pointBorderWidth: 1.5,
                    order: 2
                },
                // Actual historical data (solid, different color)
                {
                    label: 'Actual',
                    data: actualData,
                    borderColor: '#ec4899',
                    backgroundColor: 'transparent',
                    fill: false,
                    tension: 0.3,
                    borderWidth: 2,
                    pointRadius: 3,
                    pointBackgroundColor: '#f472b6',
                    pointBorderColor: '#fff',
                    pointBorderWidth: 1.5,
                    order: 1
                }
            ]
        },
        options: {
            responsive: true,
            maintainAspectRatio: false,
            interaction: {
                mode: 'index',
                intersect: false
            },
            plugins: {
                legend: {
                    display: true,
                    position: 'top',
                    labels: {
                        color: tickColor,
                        font: { family: 'Sarabun', size: 10 },
                        boxWidth: 18,
                        padding: 12,
                        usePointStyle: true
                    }
                },
                tooltip: {
                    backgroundColor: tooltipBg,
                    titleColor: tooltipTxt,
                    bodyColor: tooltipTxt,
                    borderColor: 'rgba(139, 92, 246, 0.25)',
                    borderWidth: 1,
                    padding: 10,
                    callbacks: {
                        label: function(context) {
                            const v = context.raw;
                            if (v == null) return context.dataset.label + ': N/A';
                            return context.dataset.label + ': ' + v + '%';
                        }
                    }
                }
            },
            scales: {
                x: {
                    grid: { color: gridColor },
                    ticks: { color: tickColor, font: { family: 'Sarabun', size: 10 } }
                },
                y: {
                    grid: { color: gridColor },
                    ticks: { color: tickColor, font: { family: 'Sarabun', size: 10 } },
                    beginAtZero: false,
                    suggestedMin: 40,
                    suggestedMax: 100
                }
            }
        }
    });
};
